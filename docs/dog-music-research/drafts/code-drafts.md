## fetch.sh — загрузка сэмплов

Скачивает файлы из списка с GitHub (sgossner/VSCO-2-CE, ветка master) в `tmp/trial/samples`.

````bash
#!/bin/bash
# Download the chosen VSCO 2 CE samples (CC0) from GitHub raw into tmp/trial/samples, keeping paths.
cd "$(dirname "$0")"
while IFS= read -r p; do
  [ -s "samples/$p" ] && continue
  mkdir -p "samples/$(dirname "$p")"
  u="https://raw.githubusercontent.com/sgossner/VSCO-2-CE/master/$(python3 -c 'import sys,urllib.parse;print(urllib.parse.quote(sys.argv[1]))' "$p")"
  curl -sfL --retry 3 -o "samples/$p" "$u" || echo "FAIL $p"
done < want.txt
echo FETCHDONE
````

## pitchcheck.py — проверка высоты сэмплов

Определяет основной тон файла; так проверены названия нот.

````python
"""Estimate the fundamental of sample files (harmonic product spectrum) to confirm their note mapping."""
import sys, numpy as np
from scipy.io import wavfile
def f0(path):
    sr, x = wavfile.read(path); x = x.astype(np.float64)
    if x.ndim > 1: x = x.mean(1)
    i = int(np.argmax(np.abs(x) > 0.05 * np.abs(x).max())); seg = x[i + int(0.05 * sr): i + int(1.05 * sr)]
    seg = seg * np.hanning(len(seg)); n = 1 << 20
    S = np.abs(np.fft.rfft(seg, n)); hps = S.copy()
    for h in (2, 3, 4):
        d = S[::h]; hps[:len(d)] *= d
    fr = np.fft.rfftfreq(n, 1 / sr); m = (fr > 25) & (fr < 2500)
    f = fr[m][np.argmax(hps[m])]; return sr, x.shape, 69 + 12 * np.log2(f / 440)
for p in sys.argv[1:]:
    sr, sh, midi = f0(p); print(f'{p.split("/")[-1]:40s} sr {sr} len {sh[0]/sr:5.1f}s midi {midi:6.2f}')
````

## doi_check.py — проверка DOI

Сверяет название и год каждой работы с Crossref.

````python
"""Check every DOI in a studies CSV against Crossref: title and year must match. Run: python3 -I doi_check.py <csv>"""
import csv, sys, json, urllib.request, urllib.parse, re, time
def norm(s): return re.sub(r'[^a-z0-9]', '', s.lower())
R = list(csv.DictReader(open(sys.argv[1])))
bad = 0
for r in R:
    d = (r.get('doi') or '').strip()
    if not d or not d.startswith('10.'):
        print('NO-DOI', r['year'], r['title'][:70], '|', r.get('url', '')[:80]); continue
    try:
        req = urllib.request.Request('https://api.crossref.org/works/' + urllib.parse.quote(d), headers={'User-Agent': 'doi-check/1.0 (research)'})
        m = json.load(urllib.request.urlopen(req, timeout=30))['message']
        t = (m.get('title') or [''])[0]; y = (m.get('issued', {}).get('date-parts') or [[None]])[0][0]
        ok_t = norm(t)[:40] == norm(r['title'])[:40] or norm(r['title'])[:30] in norm(t) or norm(t)[:30] in norm(r['title'])
        ok_y = str(y) == str(r['year']).strip()[:4] or abs(int(y or 0) - int(str(r['year'])[:4] or 0)) <= 1
        if not (ok_t and ok_y): bad += 1; print('MISMATCH', d, '| csv:', r['year'], r['title'][:60], '| crossref:', y, t[:60])
    except Exception as e:
        bad += 1; print('ERR', d, e)
    time.sleep(0.3)
print('rows', len(R), 'problems', bad)
````
