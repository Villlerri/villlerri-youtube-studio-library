import { chromium } from '/srv/projects/youtube-studio-library/worktrees/youtube-5-work/tmp/kit/node_modules/playwright/index.mjs';
const b = await chromium.launch({ executablePath: '/home/boardagent/.cache/ms-playwright/chromium-1248/chrome-linux64/chrome' });
for (const L of process.argv.slice(2)) {
  const p = await b.newPage({ viewport: { width: 1920, height: 1080 } });
  await p.goto('file://' + process.cwd() + '/' + L + '.html', { waitUntil: 'load' });
  await p.evaluate(() => document.fonts.ready); await p.waitForTimeout(1500);
  await p.screenshot({ path: `snap-${L}.png`, fullPage: true });
  const ov = await p.evaluate(() => document.documentElement.scrollWidth); console.log(L, 'scrollWidth', ov, 'height', await p.evaluate(() => document.documentElement.scrollHeight));
  await p.close();
}
await b.close();
